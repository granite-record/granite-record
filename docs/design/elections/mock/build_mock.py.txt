def fmt(n):
    return f"{n:,}"


def pc(n, tot):
    return f"{100*n/tot:.1f}%" if tot else "0%"


def chamber(name, seats, parts, open_=0, open_word=""):
    maj = seats // 2 + 1
    segs = "".join(f'<i class="{p}" style="flex:{n}"></i>' for n, p, _ in parts if n)
    if open_:
        segs += f'<i class="er-open" style="flex:{open_}"></i>'
    key = "".join(f'<li><span class="l-sw {p}" aria-hidden="true"></span><span><b>{n}</b> {w}</span></li>'
                  for n, p, w in parts)
    if open_:
        key += f'<li><span class="l-sw er-open" aria-hidden="true"></span><span><b>{open_}</b> {open_word}</span></li>'
    return (f'<section class="er-chb"><h3>{name}</h3><p class="er-meta">{seats} seats; {maj} make a majority.</p>'
            f'<figure class="l-fig l-bar" role="img" aria-label="{name}: '
            + "; ".join(f"{n} {w}" for n, p, w in parts) + (f"; {open_} {open_word}" if open_ else "") + '">'
            f'<div class="l-barb er-maj" style="--at:{100*maj/seats:.2f}%" aria-hidden="true">{segs}</div>'
            f'<p class="er-majw">The line is the majority, {maj} seats.</p>'
            f'<ul class="l-barkey">{key}</ul></figure></section>')


def pchip(name, party, slug=""):
    p = party.upper()
    code = p if len(p) == 1 else "X"
    inner = f'<a href="#">{esc(name)}</a>' if slug else esc(name)
    return f'<span class="mchip p-{code}" translate="no">{inner} <span class="mtag">({esc(p)})</span></span>'


def race_card(title, meta, cands, seats, foot, won_word="Elected", towns=None, linked=(), mark=True):
    tot = sum(c[2] for c in cands)
    order = sorted(cands, key=lambda c: -c[2])
    top = max(c[2] for c in cands) or 1
    rows = []
    for i, (n, p, v) in enumerate(order):
        win = mark and i < seats
        rows.append('<div class="er-row"><span>' + pchip(n, p, n in linked)
                    + (f'<span class="er-win">{won_word}</span>' if win and won_word else "")
                    + f'</span><span class="c">{fmt(v)} <span>· {pc(v, tot)}</span></span>'
                    + f'<span class="er-b" aria-hidden="true"><i class="{p.upper()[:1]}" style="width:{100*v/top:.1f}%"></i></span></div>')
    t = ""
    if towns:
        hdr, body = towns
        t = ('<details><summary>By town and ward</summary><div class="er-tw"><table class="votes"><thead><tr>'
             + "".join(f"<th>{esc(h)}</th>" for h in hdr) + "</tr></thead><tbody>"
             + "".join("<tr>" + "".join((f'<td class="n">{fmt(x)}</td>' if isinstance(x, int) else f"<td>{esc(x)}</td>") for x in r) + "</tr>" for r in body)
             + "</tbody></table></div></details>")
    return (f'<article class="er-card"><h3>{esc(title)}</h3><p class="er-meta">{meta}</p>'
            + "".join(rows) + t + (f'<p class="er-foot">{foot}</p>' if foot else "") + "</article>")


def page(name, body):
    (OUT / name).write_text(head + CSS + '<main class="wrap wide" id="main">' + body + "</main>" + tail, encoding="utf-8")

def wcands(r):
    return [(c["full_name"], c["party_abbreviation"], c.get("votes_total") or 0) for c in r["candidates"]]


def tally(files):
    done, open_ = {"R": 0, "D": 0, "I": 0}, 0
    for f in files:
        for r in json.loads((FEEDS / f).read_text())["results"]:
            seats = r["seats"] or 1
            if r["precincts_percentage"] == 100:
                for c in sorted(r["candidates"], key=lambda c: -(c.get("votes_total") or 0))[:seats]:
                    done[c["party_abbreviation"] if c["party_abbreviation"] in done else "I"] += 1
            else:
                open_ += seats
    return done, open_
